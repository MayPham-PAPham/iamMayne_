Cách 2: Kết nối tự động với Google Sheets

Dán URL Web App tại đây. Giáo viên dùng URL này để xem bảng lớp; hãy gửi cùng URL cho học viên để các em nhập một lần ở màn hình đăng nhập:

Mở hướng dẫn và mã Google Apps Script hoàn chỉnh
  1. Tạo Google Sheet mới, mở Extensions → Apps Script.
  2. Xóa mã cũ, dán toàn bộ mã ở khung bên dưới rồi lưu.
  3. Trong Project Settings → Script properties, thêm TEACHER_PASSWORD (mật khẩu riêng) và SPREADSHEET_ID (ID nằm giữa /d/ và /edit trong URL Google Sheet).
  4. Chọn Deploy → New deployment → Web app; chạy dưới tài khoản của bạn, quyền truy cập chọn Anyone.
  5. Sao chép URL kết thúc bằng /exec, dán vào ô URL phía trên và gửi URL đó cho học viên.
function getSheet_() {
  const spreadsheetId = PropertiesService.getScriptProperties().getProperty('SPREADSHEET_ID');
  if (!spreadsheetId) throw new Error('Missing Script Property: SPREADSHEET_ID');
  const ss = SpreadsheetApp.openById(spreadsheetId);
  let sh = ss.getSheetByName('Students');
  if (!sh) {
    sh = ss.insertSheet('Students');
    sh.appendRow(['studentId','name','updatedAt','coins','hintsUsed','orderingJson','dictationJson','studentJson']);
  }
  return sh;
}

function jsonOut_(data, callback) {
  const json = JSON.stringify(data);
  if (callback && /^hsk_cb_[A-Za-z0-9_]+$/.test(callback)) {
    return ContentService.createTextOutput(callback + '(' + json + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const data = JSON.parse((e.postData && e.postData.contents) || '{}');
    if (data.action !== 'saveStudent' || !data.student || !data.student.id) {
      return jsonOut_({ok:false, error:'Invalid payload'});
    }
    const st = data.student;
    const sh = getSheet_();
    const last = sh.getLastRow();
    const id = String(st.id).slice(0, 100);
    const row = [id, String(st.name || '').slice(0, 100), String(st.updatedAt || new Date().toISOString()),
      Number(st.coins || 0), Number(st.hintsUsed || 0), JSON.stringify(st.ordering || {}),
      JSON.stringify(st.dictation || {}), JSON.stringify(st)];
    let targetRow = 0;
    if (last >= 2) {
      const found = sh.getRange(2, 1, last - 1, 1).createTextFinder(id).matchEntireCell(true).findNext();
      if (found) targetRow = found.getRow();
    }
    if (targetRow) sh.getRange(targetRow, 1, 1, row.length).setValues([row]);
    else sh.appendRow(row);
    return jsonOut_({ok:true});
  } catch (err) {
    return jsonOut_({ok:false, error:String(err)});
  }
}

function doGet(e) {
  const callback = String((e.parameter && e.parameter.callback) || '');
  try {
    const props = PropertiesService.getScriptProperties();
    const expected = props.getProperty('TEACHER_PASSWORD');
    const supplied = String((e.parameter && e.parameter.password) || '');
    if (!expected) return jsonOut_({ok:false, error:'Teacher password is not configured'}, callback);
    if (supplied !== expected) return jsonOut_({ok:false, error:'Wrong teacher password'}, callback);
    const sh = getSheet_();
    const last = sh.getLastRow();
    const students = [];
    if (last >= 2) {
      const values = sh.getRange(2, 1, last - 1, 8).getValues();
      values.forEach(row => {
        try {
          const st = JSON.parse(String(row[7] || '{}'));
          st.id = String(row[0] || st.id);
          st.name = String(row[1] || st.name || 'Học viên');
          st.updatedAt = String(row[2] || st.updatedAt || '');
          st.coins = Number(row[3] || st.coins || 0);
          st.hintsUsed = Number(row[4] || st.hintsUsed || 0);
          st.ordering = st.ordering || JSON.parse(String(row[5] || '{}'));
          st.dictation = st.dictation || JSON.parse(String(row[6] || '{}'));
          st.isCloud = true;
          students.push(st);
        } catch (parseErr) {}
      });
    }
    return jsonOut_({ok:true, students:students}, callback);
  } catch (err) {
    return jsonOut_({ok:false, error:String(err)}, callback);
  }
}
